quasarcraft_spawn = {
	set_global_flag = giga_quasarcraft_constructed
	create_fleet = {
		name = "The Gargantuan Quasarcraft"
		settings = {
			spawn_debris = no
			can_change_composition = no
			is_ultra_boss = yes
		}
		effect = {
			set_fleet_flag = quasarcraft_fleet
			set_owner = prev
			create_ship = {
				name = "The Gargantuan Quasarcraft"
				random_existing_design = carks_quasark
				graphical_culture = none
			}
			#set_location = $LOCATION$
		}
	}
}
